    if(portable_sdk_esp_dsp)
        list(APPEND sdk_dependencies esp-dsp)

if(NOT DEFINED PORTABLE_SDK_MAX_FFT_SIZE)
    if(DEFINED CONFIG_PORTABLE_SDK_MAX_FFT_SIZE)
        set(PORTABLE_SDK_MAX_FFT_SIZE "${CONFIG_PORTABLE_SDK_MAX_FFT_SIZE}")
    else()
        set(PORTABLE_SDK_MAX_FFT_SIZE 4096)
    endif()
endif()
if(NOT PORTABLE_SDK_MAX_FFT_SIZE MATCHES "^(512|1024|2048|4096|8192|16384)$")
    message(FATAL_ERROR "PORTABLE_SDK_MAX_FFT_SIZE must be 512, 1024, 2048, 4096, 8192 or 16384")
endif()
if(PORTABLE_SDK_ESP_DSP AND DEFINED CONFIG_DSP_MAX_FFT_SIZE AND
   CONFIG_DSP_MAX_FFT_SIZE LESS PORTABLE_SDK_MAX_FFT_SIZE)
    message(FATAL_ERROR "ESP-DSP CONFIG_DSP_MAX_FFT_SIZE must cover PORTABLE_SDK_MAX_FFT_SIZE")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/sources.cmake")
set(sdk_dependencies bdsp)
if(PORTABLE_SDK_ESP_DSP)
    list(APPEND sdk_dependencies esp-dsp)
endif()
idf_component_register(SRCS ${PORTABLE_SDK_SOURCES}
    INCLUDE_DIRS ${PORTABLE_SDK_INCLUDES} REQUIRES ${sdk_dependencies})
set_target_properties(${COMPONENT_LIB} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_compile_definitions(${COMPONENT_LIB} PUBLIC PORTABLE_DSP_MAX_FFT_SIZE=${PORTABLE_SDK_MAX_FFT_SIZE})
target_compile_definitions(${COMPONENT_LIB} PUBLIC DSP_FFT_USE_C_ALIGNED_STORAGE=1)
option(PORTABLE_SDK_PLATFORM_RFFT "Include Xtensa RFFT kernels in IRAM" ON)
option(PORTABLE_SDK_RFFT_EXTRA_OPTIMIZATIONS "Include contiguous input fusion and specialized radix2 kernel" ON)
if(PORTABLE_SDK_ESP_DSP AND (CONFIG_IDF_TARGET_ESP32 OR CONFIG_IDF_TARGET_ESP32S3) AND CONFIG_DSP_OPTIMIZED AND PORTABLE_SDK_PLATFORM_RFFT)
    option(PORTABLE_SDK_FUSED_RFFT "Fuse mixed-radix and real reconstruction on Xtensa" ON)
    target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fft_ae32.S")
    if(CONFIG_IDF_TARGET_ESP32S3)
        target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fft4r_rfft_aes3.S")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE DSP_FFT_AES3=1)
        if(PORTABLE_SDK_RFFT_EXTRA_OPTIMIZATIONS)
            target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fft2r_rfft_aes3.S")
        endif()
    else()
        target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fft4r_rfft_ae32.S")
        if(PORTABLE_SDK_RFFT_EXTRA_OPTIMIZATIONS)
            target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fft2r_rfft_ae32.S")
        endif()
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE DSP_FFT_AE32=1)
    if(PORTABLE_SDK_RFFT_EXTRA_OPTIMIZATIONS)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE DSP_FFT_EXTRA=1)
    endif()
    if(PORTABLE_SDK_FUSED_RFFT)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE DSP_FFT_FUSED=1)
    endif()
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=gnu++20;-fno-exceptions;-fno-rtti>)

    # The vendored portable dsp_common header expects the IDF version macros
    # when __XTENSA__ is defined; provide the platform header at this boundary.
    target_compile_options(${COMPONENT_LIB} PRIVATE -include esp_idf_version.h)
endif()
