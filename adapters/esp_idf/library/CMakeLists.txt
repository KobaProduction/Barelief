cmake_minimum_required(VERSION 3.30)
# This composition root builds IDF component archives, never a firmware ELF.
set(PORTABLE_SDK_MAX_FFT_SIZE 4096 CACHE STRING "Compiled FFT capacity")
option(PORTABLE_SDK_ESP_DSP "Use optimized ESP-DSP" ON)
option(PORTABLE_SDK_TRANSPORTS "Build TCP/UART platform adapter" ON)
set(ESP32_TOOLCHAIN_DIR "$ENV{ESP32_TOOLCHAIN_DIR}" CACHE PATH "Existing compiler root (contains bin)")
set(ESP_IDF_SOURCE_DIR "$ENV{IDF_PATH}" CACHE PATH "Existing ESP-IDF sources")
if(NOT ESP_IDF_SOURCE_DIR AND DEFINED ENV{IDF_PATH})
    set(ESP_IDF_SOURCE_DIR "$ENV{IDF_PATH}" CACHE PATH "Existing ESP-IDF sources" FORCE)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/preflight.cmake")
portable_esp32_preflight()
include(FetchContent)
if(NOT ESP_IDF_SOURCE_DIR)
    message(FATAL_ERROR "Provide existing ESP-IDF through IDF_PATH or ESP_IDF_SOURCE_DIR; this build never installs a platform")
endif()
if(NOT EXISTS "${ESP_IDF_SOURCE_DIR}/tools/cmake/toolchain-esp32.cmake")
    message(FATAL_ERROR "ESP_IDF_SOURCE_DIR must contain ESP-IDF sources")
endif()
set(ENV{IDF_PATH} "${ESP_IDF_SOURCE_DIR}")
set(ENV{IDF_COMPONENT_MANAGER} 0)
set(IDF_TARGET esp32)
if(NOT EXISTS "${ESP_IDF_SOURCE_DIR}/tools/cmakev2/idf.cmake")
    message(FATAL_ERROR "The library-only target requires ESP-IDF 6.1 with its native CMake v2 library API")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_BINARY_DIR}/sdkconfig.defaults")
file(WRITE "${SDKCONFIG_DEFAULTS}"
    "CONFIG_PORTABLE_SDK_FFT_CAPACITY_${PORTABLE_SDK_MAX_FFT_SIZE}=y\nCONFIG_COMPILER_OPTIMIZATION_PERF=y\n")
if(PORTABLE_SDK_ESP_DSP)
    file(APPEND "${SDKCONFIG_DEFAULTS}"
        "CONFIG_DSP_MAX_FFT_SIZE_${PORTABLE_SDK_MAX_FFT_SIZE}=y\nCONFIG_DSP_OPTIMIZED=y\n")
endif()
include("${ESP_IDF_SOURCE_DIR}/tools/cmakev2/idf.cmake")
project(PortableDSP_ESP32 VERSION 1.0.0 LANGUAGES C CXX ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT BDSP_SOURCE_DIR)
    FetchContent_Declare(esp32_bdsp
        GIT_REPOSITORY https://github.com/KobaProduction/BDSP.git
        GIT_TAG 2c68056dfaeffba2f34fc64eee9bc9e331e83c7c
        SOURCE_SUBDIR source-only)
    FetchContent_MakeAvailable(esp32_bdsp)
    set(BDSP_SOURCE_DIR "${esp32_bdsp_SOURCE_DIR}" CACHE PATH "BDSP sources" FORCE)
endif()
if(PORTABLE_SDK_ESP_DSP)
    FetchContent_Declare(esp_dsp
        GIT_REPOSITORY https://github.com/espressif/esp-dsp.git
        GIT_TAG a53a0756833c045311ea1d79a2badf495cdfde4c
        SOURCE_DIR "${FETCHCONTENT_BASE_DIR}/esp-dsp"
        SOURCE_SUBDIR source-only)
    FetchContent_MakeAvailable(esp_dsp)
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../bdsp" "${CMAKE_CURRENT_LIST_DIR}/../portable_sdk")
set(sdk_components portable_sdk bdsp)
if(PORTABLE_SDK_ESP_DSP)
    list(APPEND EXTRA_COMPONENT_DIRS "${esp_dsp_SOURCE_DIR}")
    list(APPEND sdk_components esp-dsp)
endif()
if(PORTABLE_SDK_TRANSPORTS)
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../stream_adapter")
    list(APPEND sdk_components stream_adapter)
endif()
idf_project_init()
idf_build_library(portable_esp32_components COMPONENTS ${sdk_components})
add_custom_target(esp32_sdk)
foreach(component IN LISTS sdk_components)
    idf_component_get_property(component_lib "${component}" COMPONENT_TARGET)
    add_dependencies(esp32_sdk ${component_lib})
    install(TARGETS ${component_lib} ARCHIVE DESTINATION lib/esp32)
endforeach()
get_filename_component(sdk_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(part IN ITEMS libs/signal_contracts libs/bdsp_protocol domain application
    libs/dataflow_runtime libs/dsp_kernels libs/embedded_support adapters/esp_idf/stream_adapter)
    install(DIRECTORY "${sdk_root}/${part}/include/" DESTINATION include)
endforeach()
install(DIRECTORY "${BDSP_SOURCE_DIR}/include/" DESTINATION include)
install(FILES "${CMAKE_BINARY_DIR}/sdkconfig" "${CMAKE_BINARY_DIR}/config/sdkconfig.h"
    DESTINATION share/PortableDSP/esp32)
idf_build_get_property(profile_idf_version IDF_VER)
file(WRITE "${CMAKE_BINARY_DIR}/build-profile.txt"
    "target=esp32\ncompiler=${CMAKE_CXX_COMPILER}\ncompiler_version=${CMAKE_CXX_COMPILER_VERSION}\nidf=${profile_idf_version}\nfft_capacity=${PORTABLE_SDK_MAX_FFT_SIZE}\nesp_dsp=${PORTABLE_SDK_ESP_DSP}\ntransports=${PORTABLE_SDK_TRANSPORTS}\n")
install(FILES "${CMAKE_BINARY_DIR}/build-profile.txt" DESTINATION share/PortableDSP/esp32)
