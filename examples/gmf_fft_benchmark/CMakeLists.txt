cmake_minimum_required(VERSION 3.24)
option(GMF_SPECTRUM_ONLY "Export exact per-bin diagnostic data without timing sweep" OFF)
option(GMF_EXTENDED_SWEEP "Measure ten routes, core/full/inverse and RAM through16384" OFF)
option(GMF_R4_S3_SIMD "Experimental PIE radix4 backend; hardware acceptance pending" OFF)
option(GMF_R4_STAGE_PROFILE "Instrument experimental R4 phases; not a performance image" OFF)
option(GMF_BENCHMARK_4096_ONLY "Bound the performance sweep to4096 points" OFF)
option(GMF_S3_VECTOR_PERMUTATION "Experimental exact S3 permutation for complex lengths>1024" OFF)
set(PORTABLE_SDK_ESP_DSP ON)
if(GMF_EXTENDED_SWEEP)
    set(PORTABLE_SDK_MAX_FFT_SIZE 16384)
else()
    set(PORTABLE_SDK_MAX_FFT_SIZE 4096)
endif()
if(NOT DEFINED ENV{IDF_PATH} OR NOT EXISTS "${ESP_DSP_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Provide existing IDF_PATH and ESP_DSP_SOURCE_DIR")
endif()
set(EXTRA_COMPONENT_DIRS
    "${CMAKE_CURRENT_LIST_DIR}/../../adapters/esp_idf/bdsp"
    "${CMAKE_CURRENT_LIST_DIR}/../../adapters/esp_idf/portable_sdk"
    "${ESP_DSP_SOURCE_DIR}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(GMF_R4_S3_SIMD AND (NOT IDF_TARGET STREQUAL "esp32s3" OR NOT GMF_EXTENDED_SWEEP))
    message(FATAL_ERROR "GMF_R4_S3_SIMD requires ESP32-S3 and GMF_EXTENDED_SWEEP")
endif()
if(GMF_R4_STAGE_PROFILE AND NOT GMF_R4_S3_SIMD)
    message(FATAL_ERROR "GMF_R4_STAGE_PROFILE requires GMF_R4_S3_SIMD")
endif()
if(GMF_S3_VECTOR_PERMUTATION AND (NOT IDF_TARGET STREQUAL "esp32s3" OR NOT GMF_R4_S3_SIMD))
    message(FATAL_ERROR "GMF_S3_VECTOR_PERMUTATION requires the S3 SIMD hardware preflight build")
endif()
idf_build_set_property(PORTABLE_SDK_ESP_DSP ON)
idf_build_set_property(GMF_FFT_SOURCE_DIR "${GMF_FFT_SOURCE_DIR}")
idf_build_set_property(GMF_R4_S3_SIMD "${GMF_R4_S3_SIMD}")
idf_build_set_property(GMF_R4_STAGE_PROFILE "${GMF_R4_STAGE_PROFILE}")
set(COMPONENTS main)
project(gmf_fft_benchmark)
