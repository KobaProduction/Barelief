add_subdirectory(vendor/esp_dsp)
set(PORTABLE_DSP_MAX_FFT_SIZE 4096 CACHE STRING "Maximum portable FFT size (512, 1024, 2048, 4096)")
set_property(CACHE PORTABLE_DSP_MAX_FFT_SIZE PROPERTY STRINGS 512 1024 2048 4096)
if(NOT PORTABLE_DSP_MAX_FFT_SIZE MATCHES "^(512|1024|2048|4096)$")
    message(FATAL_ERROR "PORTABLE_DSP_MAX_FFT_SIZE must be 512, 1024, 2048 or 4096")
endif()
add_library(portable_dsp STATIC src/dsp.cpp)
target_compile_definitions(portable_dsp PUBLIC PORTABLE_DSP_MAX_FFT_SIZE=${PORTABLE_DSP_MAX_FFT_SIZE})
target_include_directories(portable_dsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(portable_dsp PUBLIC cxx_std_20)
target_link_libraries(portable_dsp PRIVATE dsp)
