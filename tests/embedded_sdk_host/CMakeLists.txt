cmake_minimum_required(VERSION 3.30)
project(embedded_sdk_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)
set(PORTABLE_DISABLE_EXCEPTIONS_RTTI ON)
set(PORTABLE_DSP_MAX_FFT_SIZE 512 CACHE STRING "")
set(BDSP_INSTALL OFF CACHE BOOL "")
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT BDSP_SOURCE_DIR)
    message(FATAL_ERROR "Pass pinned BDSP_SOURCE_DIR explicitly")
endif()
add_subdirectory("${BDSP_SOURCE_DIR}" bdsp)
foreach(part IN ITEMS libs/signal_contracts libs/wled_protocol libs/bdsp_protocol domain application adapters/control libs/dsp_kernels libs/dataflow_runtime libs/embedded_support)
    add_subdirectory("${SDK_ROOT}/${part}" "${part}")
endforeach()
foreach(target IN ITEMS BDSP signal_contracts bdsp_protocol graph_domain graph_application dsp dsp_kernels dataflow_runtime embedded_support)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
endforeach()
enable_testing()
include("${SDK_ROOT}/tests/embedded_sdk_tests.cmake")
include("${SDK_ROOT}/adapters/esp_idf/install_sources.cmake")
