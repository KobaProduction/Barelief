if(NOT SDK_ROOT)
    get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
add_library(fake_idf INTERFACE)
target_include_directories(fake_idf INTERFACE "${CMAKE_CURRENT_LIST_DIR}/fake_idf")
function(idf_component_register)
    cmake_parse_arguments(IDF "" "" "SRCS;INCLUDE_DIRS;REQUIRES" ${ARGN})
    foreach(source IN LISTS IDF_SRCS)
        if(NOT EXISTS "${source}" AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
            message(FATAL_ERROR "Missing component source: ${source}")
        endif()
    endforeach()
    set(target "smoke_${SMOKE_COMPONENT}")
    add_library(${target} STATIC ${IDF_SRCS})
    target_include_directories(${target} PUBLIC ${IDF_INCLUDE_DIRS})
    target_link_libraries(${target} PUBLIC fake_idf)
    foreach(dependency IN LISTS IDF_REQUIRES)
        if(TARGET smoke_${dependency})
            target_link_libraries(${target} PUBLIC smoke_${dependency})
        elseif(NOT dependency MATCHES "^(esp_driver_uart|lwip|freertos|esp_timer)$")
            message(FATAL_ERROR "Missing explicit component dependency: ${dependency}")
        endif()
    endforeach()
    set(COMPONENT_LIB ${target} PARENT_SCOPE)
endfunction()
