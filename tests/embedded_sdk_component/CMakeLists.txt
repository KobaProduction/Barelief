function(idf_component_register)
    cmake_parse_arguments(IDF "" "" "SRCS;INCLUDE_DIRS;REQUIRES" ${ARGN})
    foreach(source IN LISTS IDF_SRCS)
        if(NOT EXISTS "${source}" AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
            message(FATAL_ERROR "Missing component source: ${source}")
        endif()
    endforeach()
    set(target "smoke_${SMOKE_COMPONENT}")
    add_library(${target} STATIC ${IDF_SRCS})
    target_include_directories(${target} PUBLIC ${IDF_INCLUDE_DIRS})
    target_link_libraries(${target} PUBLIC fake_idf)
    foreach(dependency IN LISTS IDF_REQUIRES)
        if(TARGET smoke_${dependency})
            target_link_libraries(${target} PUBLIC smoke_${dependency})
        elseif(NOT dependency MATCHES "^(esp_driver_uart|lwip|freertos|esp_timer|nvs_flash)$")
            message(FATAL_ERROR "Missing explicit component dependency: ${dependency}")
        endif()
    endforeach()
    set(COMPONENT_LIB ${target} PARENT_SCOPE)
endfunction()

foreach(link IN ITEMS tcp uart)
    set(target numeric_ordinary2048_${link})
    add_executable(${target} numeric_bootstrap_tests.cpp)
    target_link_libraries(${target} PRIVATE smoke_stream_adapter smoke_persistence)
    target_compile_definitions(${target} PRIVATE ORDINARY_PROFILE=1 CONFIG_STREAMER_SPECTRAL=1 CONFIG_STREAMER_FFT_SIZE=2048 CONFIG_STREAMER_MEMORY_KIB=320)
    if(link STREQUAL "uart")
        target_compile_definitions(${target} PRIVATE CONFIG_STREAMER_UART=1 CONFIG_STREAMER_UART_BAUD=2000000 CONFIG_STREAMER_UART_TX=17 CONFIG_STREAMER_UART_RX=16)
    else()
        target_compile_definitions(${target} PRIVATE CONFIG_STREAMER_TCP=1 CONFIG_STREAMER_TCP_BUDGET=200000)
    endif()
    target_compile_options(${target} PRIVATE -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
    target_link_options(${target} PRIVATE -Wl,--gc-sections)
    # Host64 accounting intentionally rejects this 320KiB profile; native capacity is separate.
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        add_test(NAME ${target}_host_budget_refusal COMMAND ${target})
    else()
        add_test(NAME ${target} COMMAND ${target})
    endif()
endforeach()
