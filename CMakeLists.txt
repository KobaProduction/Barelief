include(cmake/deploy_linux_qml.cmake)
install(CODE [[get_filename_component(CMAKE_INSTALL_PREFIX "${CMAKE_INSTALL_PREFIX}" ABSOLUTE)]])
foreach(app IN ITEMS core_app qt-application pipeline_cli aggregator_server_app aggregator_client_app)
    if(NOT TARGET ${app})
        continue()
    endif()
    if(UNIX AND NOT APPLE)
        barelief_linux_install_rpath(${app})
    endif()
    install(TARGETS ${app} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(app STREQUAL "core_app" AND UNIX AND NOT APPLE)
        barelief_generate_linux_qml_deploy_script(${app} deploy_script)
    elseif(app STREQUAL "core_app")
        qt_generate_deploy_app_script(TARGET ${app} OUTPUT_SCRIPT deploy_script
            INCLUDE_PLUGINS qoffscreen
            DEPLOY_TOOL_OPTIONS --qmldir "${PROJECT_SOURCE_DIR}/presentation/qt/qml")
    else()
        qt_generate_deploy_app_script(TARGET ${app} OUTPUT_SCRIPT deploy_script)
    endif()
    install(SCRIPT ${deploy_script})
endforeach()
